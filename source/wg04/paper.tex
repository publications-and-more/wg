\section{Definitions}

\subsection{Definition of a sum of an integral from $a_{0}$ to $a_{n}$}

Let an integral $\alpha$:
\begin{equation}
    \int_{a_{0}}^{a_{n}}\sum_{b_{a_{0}},k=a_{0}} f_{b_{a_{0}}}(S(k+b_{a_{0}})) + C
\end{equation}
Such that
\begin{equation}
    a = 0, \quad a \in \mathbb{N}, \quad k \neq 0, \quad k \in \mathbb{R}
\end{equation}
Such that $S$ is a user-defined function computing a sum of $k-b_{a_{0}}$.
