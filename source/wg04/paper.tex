\section{Definitions}

We assume $x, y, z \in \mathbb{R}, \quad n \in \mathbb{N}, \quad \cos(\pi) < a < b < \cos(0), \quad t = a$
\subsection{The A-Integral:}
\begin{equation}
    I_{1} \coloneqq \sum_{n}^{t} \int_{a}^{b} S(x) dx
\end{equation}
\subsection{The A-Function:}
\begin{equation}
	S(x) : (x, y) \to \mathbb{R}
\end{equation}
\subsection{The A-Equation:}
\begin{equation}
	T_{2} \coloneqq \int_{a}^{b} E_{1}dx, \quad E_{1} = Z(x) := \frac{(S(x) \cdot y)}{\Delta t \cdot z}
\end{equation}
