\subsection{The Sum Integral:}
\begin{equation}
    \int_{a}^{b} S(x)dx, \quad \forall a, b \in \mathbb{R}, \quad 0 < a < b
\end{equation}
\subsection{The Sum Function:}
\begin{equation}
    S(x) := x^{e}, \quad x \in \mathbb{R}, \quad x > 0
\end{equation}
