\subsection{DEFINITION OF THE S-INTEGRAL:}
\begin{equation}
    I_{1}(a, b, t, u) \coloneqq \int_{a}^{b} S(t, u)dt
\end{equation}
\subsection{DEFINITION OF THE S-FUNCTION:}
\begin{equation}
    S(x, u) = \sum_{k=1}^{t} (\frac{dx}{du}\cdot k) \cdot x, \quad x > 0, \quad t > k
\end{equation}
