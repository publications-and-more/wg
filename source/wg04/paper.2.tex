\section{Definitions}

We assume $x \in \mathbb{R}, \quad a, b \in \mathbb{N}, \quad 0 < a < b < \cos(0)$ for:

\subsection{The S-Integral:}
\begin{equation}
    I_{1} \coloneqq \int_{a}^{b} S(x)dx
\end{equation}
\subsection{The S-Function:}
\begin{equation}
    S(x) = \sum_{k=1}^{t} (\frac{dx}{du}) \cdot x, \quad x > 0, \quad t > k
\end{equation}
